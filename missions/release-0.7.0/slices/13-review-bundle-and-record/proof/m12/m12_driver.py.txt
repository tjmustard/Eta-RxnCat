"""M12 on the real v0.7.0 review bundle (slice 13): Chrome and Playwright Firefox from file://.

Usage: python m12_driver_v070.py <bundle_root> <evidence_dir>
Starts at the root index.html, opens every sub-bundle by its link, and for every candidate checks
that each TS frequency's mode animates and each trajectory plays. Throwaway Playwright; bypass_csp
is never set. Reads only the scratch tree.
"""

import json
import sys
import time
from pathlib import Path

from playwright.sync_api import sync_playwright

ROOT = Path(sys.argv[1]).resolve()
OUT = Path(sys.argv[2]).resolve()
CONTROL = OUT / "control.html"

INIT = """
window.__csp = [];
document.addEventListener("securitypolicyviolation", function (e) {
  window.__csp.push(e.violatedDirective + " " + e.blockedURI);
});
"""


class Check:
    def __init__(self):
        self.log, self.failed, self.passed = [], 0, 0

    def __call__(self, name, ok, detail=""):
        self.failed += not ok
        self.passed += bool(ok)
        self.log.append(f"{'PASS' if ok else 'FAIL'}  {name}" + (f"  [{detail}]" if detail else ""))


def clean(text):
    text = text.replace(ROOT.as_uri(), "<bundle>").replace(str(ROOT), "<bundle>")
    text = text.replace(CONTROL.as_uri(), "<control>").replace(str(OUT), "<evidence>")
    return text.replace(str(Path.home()), "<home>")


def watch(page, console, network, dialogs):
    page.on("console", lambda m: console.append(f"console.{m.type}: {m.text[:300]} @ {m.location.get('url', '')}"))
    page.on("pageerror", lambda e: console.append(f"pageerror: {e}"))
    page.on("crash", lambda p: console.append("crash"))
    page.on("request", lambda r: network.append(f"request {r.resource_type} {r.method} {r.url}"))
    page.on("requestfailed", lambda r: network.append(f"requestfailed {r.url} {r.failure}"))
    page.on("websocket", lambda w: network.append(f"websocket {w.url}"))
    page.on("dialog", lambda d: (dialogs.append(d.message), d.dismiss()))


def shots_differ(page, box, n=4):
    images = set()
    for _ in range(n):
        images.add(box.screenshot())
        if len(images) > 1:
            return True, len(images)
        page.wait_for_timeout(240)
    return False, len(images)


def check_candidate(page, check, sub, data, candidate_id, shots, counts):
    section = page.locator(f'.eta-panels[data-candidate="{candidate_id}"]')
    frequencies = {f["frequency_id"]: f for f in data["frequencies"]}
    trajectories = {t["trajectory_id"]: t for t in data["trajectories"]}
    ts_ids = {i for c in data["candidates"] if c["candidate_id"] == candidate_id for i in c["frequency_ids"]}
    left = section.locator(".eta-left select")
    options = left.locator("option").evaluate_all("els => els.map(e => e.value)")
    expected_options = [
        i for c in data["candidates"] if c["candidate_id"] == candidate_id
        for i in [*c["frequency_ids"], *c["endpoint_frequency_ids"]]
    ]
    check(f"{sub} {candidate_id}: frequency options are the record's", options == expected_options,
          f"{len(options)} vs {len(expected_options)}")
    kept = set()
    for option in options:
        if option not in ts_ids:
            continue
        entry = frequencies[option]
        left.select_option(option)
        page.wait_for_timeout(250)
        name = f"{sub} TS {entry['species_id']} {entry['stage_key'][:12]}"
        if entry["status"] != "completed":
            check(f"{name}: frequencies not completed, warning shown",
                  section.locator(".eta-left .warn").count() > 0, entry["status"])
            counts["ts_not_completed"] += 1
            continue
        counts["ts"] += 1
        readout = section.locator(".eta-left .eta-mode").inner_text()
        if entry["n_imaginary"] == 0:
            check(f"{name}: no imaginary mode stated", "No imaginary mode." in readout)
            continue
        box = section.locator(".eta-left .eta-3d")
        box.scroll_into_view_if_needed()
        ok, n = shots_differ(page, box)
        counts["animated"] += ok
        mode = entry["imaginary_modes"][0]["frequency_cm1"]
        check(f"{name}: mode animates ({entry['n_imaginary']} imaginary, {mode:.1f} cm-1)",
              ok and f"{mode:.1f}" in readout, f"{n} distinct")
        unit = entry["species_id"]
        if unit not in kept:
            kept.add(unit)
            box.screenshot(path=str(shots / f"{sub}-ts-{unit}.png"))
    right = section.locator(".eta-right select")
    options = right.locator("option").evaluate_all("els => els.map(e => e.value)")
    expected_options = [
        i for c in data["candidates"] if c["candidate_id"] == candidate_id
        for i in [*c["trajectory_ids"], *c["endpoint_trajectory_ids"]]
    ]
    check(f"{sub} {candidate_id}: trajectory options are the record's", options == expected_options,
          f"{len(options)} vs {len(expected_options)}")
    for option in options:
        entry = trajectories[option]
        frames = len(entry["frames"])
        right.select_option(option)
        player = section.locator(".eta-player")
        name = f"{sub} {entry['kind']} {entry['species_id']} {entry['stage_key'][:12]} ({frames} frames, stride {entry['decimation_stride']})"
        readout = player.locator(".eta-readout")
        first = readout.inner_text()
        moved = True
        if frames > 1:
            player.get_by_role("button", name="Play").click()
            try:
                page.wait_for_function(
                    "([s, t]) => document.querySelector(s).textContent.trim() !== t.trim()",
                    arg=[f'.eta-panels[data-candidate="{candidate_id}"] .eta-player .eta-readout', first],
                    timeout=4000,
                )
            except Exception as exc:  # noqa: BLE001
                moved = False
                console_note = str(exc)[:100]
        slider = player.locator("input[type=range]")
        slider.fill(str(frames - 1))
        last = readout.inner_text()
        marker = section.locator(".eta-current").get_attribute("data-step")
        ok = moved and f"frame {frames} of {frames}" in last and marker == str(frames - 1)
        stated = entry["decimation_stride"] == 1 or "decimated: every" in player.inner_text()
        counts["played"] += ok
        counts["trajectories"] += 1
        check(f"{name}: plays and reaches the last frame, marker follows", ok and stated, last.strip()[:60])
    page.screenshot(path=str(shots / f"{sub}-panels.png"))


def run(browser_name, browser):
    shots = OUT / browser_name
    shots.mkdir(parents=True, exist_ok=True)
    check = Check()
    console, network, dialogs = [], [], []
    counts = {"ts": 0, "animated": 0, "ts_not_completed": 0, "trajectories": 0, "played": 0, "pages": 0}
    context = browser.new_context(viewport={"width": 1400, "height": 1000})
    context.add_init_script(INIT)
    page = context.new_page()
    watch(page, console, network, dialogs)
    start = time.time()
    page.goto((ROOT / "index.html").as_uri(), wait_until="load")
    page.wait_for_timeout(500)
    check("root opened from file://", page.url.startswith("file://"))
    check("root has no script", page.evaluate("document.scripts.length") == 0)
    page.screenshot(path=str(shots / "00-root-index.png"), full_page=True)
    links = page.locator("tbody a[href$='/index.html']").evaluate_all("els => els.map(e => e.getAttribute('href'))")
    subs = sorted(p.name for p in ROOT.iterdir() if p.is_dir())
    check("one root link per sub-bundle", sorted(h.split("/")[0] for h in links) == subs, f"{len(links)} links")
    violations = []
    for href in links:
        sub = href.split("/")[0]
        page.goto((ROOT / "index.html").as_uri(), wait_until="load")
        page.locator(f'a[href="{href}"]').click()
        page.wait_for_load_state("load")
        page.wait_for_timeout(2000)
        counts["pages"] += 1
        data = json.loads((ROOT / sub / "run_data.json").read_bytes())
        check(f"{sub}: link opens the sub-bundle", page.url.endswith(f"{sub}/index.html"))
        check(f"{sub}: no load error", page.locator("#eta-load-error").count() == 0)
        check(f"{sub}: candidate table lists every entry",
              page.locator(".eta-candidates tbody tr").count() == len(data["candidates"]))
        page.screenshot(path=str(shots / f"{sub}-page.png"))
        for candidate_id in dict.fromkeys(c["candidate_id"] for c in data["candidates"]):
            page.locator(".eta-candidates a", has_text=candidate_id).first.click()
            page.wait_for_timeout(1000)
            check(f"{sub} {candidate_id}: index link mounts the panels",
                  page.locator(f'.eta-panels[data-candidate="{candidate_id}"] canvas').count() == 2)
            check_candidate(page, check, sub, data, candidate_id, shots, counts)
        violations += page.evaluate("window.__csp")
        print(f"{browser_name} {sub} done at {time.time() - start:.0f}s, failed so far {check.failed}", flush=True)
    check("no CSP violation", violations == [], str(violations[:5]))
    check("no dialog", dialogs == [], str(dialogs))
    context.close()

    context = browser.new_context()
    context.add_init_script(INIT)
    page = context.new_page()
    page.goto((ROOT / subs[0] / "index.html").as_uri(), wait_until="load")
    page.evaluate("""() => {
      var img = document.createElement("img"); img.src = "http://127.0.0.1:59999/m12-csp-control";
      document.body.appendChild(img);
    }""")
    page.wait_for_timeout(700)
    blocked = page.evaluate("window.__csp")
    check("control: the page's policy blocks a remote image and the listener reports it",
          any(v.startswith("img-src ") for v in blocked), str(blocked))
    context.close()

    c_console, c_network = [], []
    context = browser.new_context()
    page = context.new_page()
    watch(page, c_console, c_network, [])
    CONTROL.write_text(
        '<!DOCTYPE html><title>control</title><img src="http://127.0.0.1:59999/m12-control">'
        '<script>console.error("m12-control-error")</script>'
    )
    page.goto(CONTROL.as_uri(), wait_until="load")
    page.wait_for_timeout(1500)
    context.close()

    bad_console = [c for c in console if c.startswith(("console.error", "pageerror", "crash"))]
    remote = [
        n for n in network
        if not (n.startswith("request ") and n.split(" ", 3)[3].startswith(("file:", "blob:", "data:")))
    ]
    check("no console error", bad_console == [], str(bad_console[:5]))
    check("no network request (nothing but file:, blob:, data:; no failure)", remote == [], str(remote[:5]))
    check("control: the console log catches an error", any("m12-control-error" in c for c in c_console))
    check("control: the network log catches a request", any("http://127.0.0.1:59999/m12-control" in n for n in c_network))

    (shots / "console.txt").write_text(clean("\n".join(console) + "\n" if console else "(no console message, no page error)\n"))
    (shots / "network.txt").write_text(clean("\n".join(network) + "\n" if network else "(no request event)\n"))
    (shots / "control-logs.txt").write_text(clean("\n".join(c_console + c_network) + "\n"))
    summary = f"{browser_name} {browser.version}\ncounts: {counts}\npassed: {check.passed} failed: {check.failed} in {time.time() - start:.0f}s\n"
    (shots / "checks.txt").write_text(clean(summary + "\n".join(check.log) + "\n"))
    print(summary)
    for line in check.log:
        if line.startswith("FAIL"):
            print(clean(line))
    return check.failed


with sync_playwright() as p:
    failed = 0
    chrome = p.chromium.launch(channel="chrome", args=["--use-angle=swiftshader", "--enable-unsafe-swiftshader"])
    failed += run("chrome", chrome)
    chrome.close()
    firefox = p.firefox.launch(firefox_user_prefs={"webgl.force-enabled": True})
    failed += run("firefox", firefox)
    firefox.close()
print("total failed:", failed)
sys.exit(1 if failed else 0)
